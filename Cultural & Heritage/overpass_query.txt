// ---------- QUERY 1: heritage sites  ->  save as data/sites.geojson ----------
// Finds anything tagged historic, museums, heritage-listed places,
// and cathedrals inside Butuan City.

[out:json][timeout:90];
{{geocodeArea:Butuan City, Agusan del Norte}}->.a;
(
  nwr["historic"](area.a);
  nwr["tourism"="museum"](area.a);
  nwr["heritage"](area.a);
  nwr["amenity"="place_of_worship"]["building"="cathedral"](area.a);
);
out center;


// ---------- QUERY 2: the Agusan River  ->  save as data/agusan_river.geojson ----------
// If this returns nothing, remove the ["name"~"Agusan"] filter and delete
// the extra waterways in the export, or draw the river line yourself in geojson.io.

[out:json][timeout:90];
{{geocodeArea:Butuan City, Agusan del Norte}}->.a;
(
  way["waterway"="river"]["name"~"Agusan"](area.a);
);
out geom;
